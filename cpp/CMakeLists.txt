cmake_minimum_required(VERSION 3.20)
project(yolox_bench CXX)
set(CMAKE_CXX_STANDARD 17)
# ncnn 은 제품과 같은 20260113 공식 소스, ORT 는 제품과 같은 1.17.1 공식 배포본(워크플로가 받아 둔다)
find_package(ncnn REQUIRED PATHS ${NCNN_DIR} NO_DEFAULT_PATH)
add_executable(bench bench.cpp)
target_include_directories(bench PRIVATE ${ORT_DIR}/include)
if(WIN32)
  target_link_libraries(bench PRIVATE ncnn ${ORT_DIR}/lib/onnxruntime.lib)
else()
  target_link_libraries(bench PRIVATE ncnn ${ORT_DIR}/lib/libonnxruntime${CMAKE_SHARED_LIBRARY_SUFFIX})
  set_target_properties(bench PROPERTIES BUILD_RPATH ${ORT_DIR}/lib)
endif()
